/**
 * Gateway reachability probes — the ONLY place probe evidence is written to the
 * route-health registry.
 *
 * WHY THIS EXISTS (2026-10-02): `recordRouteProbe()` was written for the UI
 * remodel and then never called by anything. The consequence was visible in the
 * route tray: every route sat at 'unknown' (grey) until a real chat turn
 * happened to exercise it. So the one always-on health signal the tray exists
 * to provide was blank exactly when the user most wants it — a cold start, or
 * right after rotating a key, when there has been no traffic yet.
 *
 * Design constraints, all deliberate:
 *  - NEVER blocks a caller. `probeRoutesIfStale()` is fire-and-forget because
 *    /api/health must answer immediately even while three gateways time out.
 *  - NEVER throws. This is diagnostics: a probe may fail, and must not take the
 *    caller down with it.
 *  - TTL-cached (`PROBE_TTL_MS`) so a polling UI cannot become a flood of
 *    gateway requests. The tray does not need second-by-second truth.
 *  - A probe can NEVER overwrite what a real user call established —
 *    `recordRouteProbe()` enforces that itself (see routeHealth.ts).
 */

/**
 * Probe an OpenAI-protocol gateway.
 *
 * Gateways differ: some expose a public /models list, others (OmniRoute, some
 * proxies) only allow authenticated generation calls. Probe /models first; if
 * it is restricted/absent, fall back to a minimal chat ping so the result
 * reflects what actually matters: can we GENERATE with this key?
 *
 * MOVED here verbatim from doctor.ts so the background probe and `--doctor`
 * share one implementation. Two copies would drift, and the whole point of
 * this module is that the tray and the doctor must agree about a gateway.
 */
export async function probeOpenAi(base: string, key: string, model: string): Promise<{ ok: boolean; detail: string }> {
  const ac = new AbortController()
  const t = setTimeout(() => ac.abort(), 2000)
  try {
    const res = await fetch(`${base}/models`, { headers: { authorization: `Bearer ${key}` }, signal: ac.signal })
    clearTimeout(t)
    if (res.ok) {
      const data = (await res.json()) as { data?: Array<{ id?: string; is_free?: boolean }> }
      const ids = (data.data ?? []).map((m) => m.id).filter(Boolean) as string[]
      const free = (data.data ?? [])
        .filter((m) => m.is_free || String(m.id).endsWith(':free'))
        .map((m) => m.id)
        .filter(Boolean) as string[]
      const freeTxt = free.length > 0 ? ` · free: ${free.slice(0, 4).join(', ')}${free.length > 4 ? '…' : ''}` : ''
      return { ok: true, detail: `${ids.length} model(s)${freeTxt}` }
    }
    if (res.status === 401 || res.status === 403 || res.status === 404 || res.status === 405) {
      // /models not usable — probe a minimal generation instead
      const ac2 = new AbortController()
      const t2 = setTimeout(() => ac2.abort(), 8000)
      const chat = await fetch(`${base}/chat/completions`, {
        method: 'POST',
        headers: { authorization: `Bearer ${key}`, 'content-type': 'application/json' },
        body: JSON.stringify({ model, messages: [{ role: 'user', content: 'Reply with exactly: OK' }], max_tokens: 6 }),
        signal: ac2.signal,
      })
      clearTimeout(t2)
      if (chat.ok) {
        const j = (await chat.json()) as { model?: string }
        return { ok: true, detail: `chat probe OK (model ${j.model ?? model}) — /models not exposed` }
      }
      const body = await chat.text().catch(() => '')
      return { ok: false, detail: `HTTP ${chat.status} on chat${body ? ` — ${body.replace(/\s+/g, ' ').slice(0, 100)}` : ''}` }
    }
    return { ok: false, detail: `HTTP ${res.status}` }
  } catch (e) {
    clearTimeout(t)
    return { ok: false, detail: e instanceof Error && e.name === 'AbortError' ? 'timeout (2s)' : 'not reachable' }
  }
}
